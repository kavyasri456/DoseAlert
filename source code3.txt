'use client';

import { useRouter } from 'next/navigation';
import { Button } from '@/components/ui/button';
import { Logo } from '@/components/icons';
import { LogOut } from 'lucide-react';
import { useMedicationStore } from '@/hooks/use-medication-store';
import { useEffect, useCallback } from 'react';
import { ReminderAlert } from '@/components/dashboard/reminder-alert';

export default function DashboardLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  const router = useRouter();
  const { medications, setActiveReminder, activeReminder, doseHistory } = useMedicationStore();

  const handleLogout = () => {
    localStorage.removeItem('dose-alert-token');
    router.replace('/login');
  };

  const checkReminders = useCallback(() => {
    if (activeReminder) return;

    const now = new Date();
    const currentTime = `${String(now.getHours()).padStart(2, '0')}:${String(now.getMinutes()).padStart(2, '0')}`;
    
    for (const med of medications) {
      if (med.time === currentTime) {
        const today = new Date();
        today.setHours(0, 0, 0, 0);

        const hasBeenHandledToday = doseHistory.some(dose => {
            const doseDate = new Date(dose.scheduledTime);
            doseDate.setHours(0, 0, 0, 0);
            return dose.medicationId === med.id && doseDate.getTime() === today.getTime();
        });

        if (!hasBeenHandledToday) {
          setActiveReminder(med);
          break;
        }
      }
    }
  }, [activeReminder, medications, doseHistory, setActiveReminder]);

  useEffect(() => {
    const intervalId = setInterval(checkReminders, 30000); // Check every 30 seconds
    checkReminders(); // Initial check
    return () => clearInterval(intervalId);
  }, [checkReminders]);


  return (
    <div className="flex min-h-screen w-full flex-col">
      <header className="sticky top-0 z-10 flex h-20 items-center gap-4 border-b bg-card px-4 md:px-6 shadow-md">
        <div className="flex items-center gap-2">
            <Logo className="h-10 w-10" />
            <h1 className="text-2xl font-bold font-headline text-foreground">DoseAlert</h1>
        </div>
        <div className="ml-auto">
          <Button variant="outline" onClick={handleLogout}>
            <LogOut className="mr-2 h-4 w-4" /> Logout
          </Button>
        </div>
      </header>
      <main className="flex-1 p-4 md:p-8 lg:p-12 bg-background">
        {children}
      </main>
      <ReminderAlert />
    </div>
  );
}
